# ------------------------------------------------------------
# REQUISITOS DE SISTEMA
# ------------------------------------------------------------
Node.js >= 20.9.0         # https://nodejs.org (versão LTS recomendada)
npm >= 10.0.0             # incluído com o Node.js
PostgreSQL >= 14          # https://www.postgresql.org (ou Neon cloud: https://neon.tech)

# ------------------------------------------------------------
# VARIÁVEIS DE AMBIENTE OBRIGATÓRIAS (.env)
# ------------------------------------------------------------
# DATABASE_URL   — connection string do PostgreSQL
#                  Ex: postgresql://usuario:senha@localhost:5432/biblioteca
# AUTH_SECRET    — chave aleatória para assinar os tokens JWT
#                  Gere com: node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"
